  3. 기존 코드를 지우고 아래 스크립트를 그대로 복사하여 붙여넣고 저장(Ctrl+S)합니다.
  4. 우측 상단 [배포] → [새 배포] 클릭 후 유형을 '웹 앱'으로 선택합니다.
  5. 액세스 권한을 '모든 사용자(Anyone)'로 설정한 뒤 배포하고 나오는 웹 앱 URL을 관리자 설정 창에 입력하세요!
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["등록일시", "학년", "반", "번호", "이름", "활동내용", "포인트", "소감기록"]);
    }
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      data.dateStr,
      data.grade,
      data.classNum,
      data.studentNum,
      data.studentName,
      Array.isArray(data.activities) ? data.activities.join(", ") : data.activities,
      data.points || 10,
      data.content
    ]);
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
