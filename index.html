Google Apps Script 배포 코드 안내

1. 구글 스프레드시트를 생성하고 [확장 프로그램] → [Apps Script]를 엽니다.

2. 아래 코드를 복사하여 붙여넣고 [배포] → [새 배포] → 유형: [웹 앱]을 선택합니다.

※ 중요: [액세스 권한이 있는 사용자]를 반드시 '모든 사용자(Anyone)'로 설정하고 배포한 URL을 본 웹앱에 등록해주세요.

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      data.dateStr,
      data.grade,
      data.classNum,
      data.studentNum,
      data.studentName,
      data.activities.join(', '),
      data.points,
      data.content
    ]);
    return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var rows = sheet.getDataRange().getValues();
    var logs = [];
    for (var i = 1; i < rows.length; i++) {
      var r = rows[i];
      if (!r[0] && !r[4]) continue;
      logs.push({
        dateStr: String(r[0]),
        grade: String(r[1]),
        classNum: String(r[2]),
        studentNum: String(r[3]),
        studentName: String(r[4]),
        activities: String(r[5] || '').split(', '),
        points: Number(r[6]) || 10,
        content: String(r[7] || '')
      });
    }
    return ContentService.createTextOutput(JSON.stringify({ result: "success", data: logs }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
